USE EmprestimoKit;
GO

IF OBJECT_ID('dbo.TBL_EMPRESTIMO_KIT', 'U') IS NOT NULL
    DROP TABLE dbo.TBL_EMPRESTIMO_KIT;
GO

CREATE TABLE dbo.TBL_EMPRESTIMO_KIT (
    ID BIGINT IDENTITY(1,1) PRIMARY KEY,
    TX_CODIGO_KIT VARCHAR(100) NOT NULL,
    TX_NOME_ALUNO VARCHAR(150) NOT NULL,
    HR_RETIRADA TIME NOT NULL,
    HR_LIMITE_DEVOLUCAO TIME NOT NULL,
    HR_DEVOLUCAO_REAL TIME NULL,
    TX_STATUS VARCHAR(30) NOT NULL
);
GO
